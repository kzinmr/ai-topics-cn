# Chatgpt Pro 用量用不完的可以开这些设置

# 开启 1M 上下文, 上下文超过 272K 时, input 按 2x 、output 按 1.5x
model_context_window = 1000000
# 950K 上下文时自动压缩
model_auto_compact_token_limit = 950000
# 1.5 倍速度, 双倍额度消耗
service_tier = "fast"
# 配合"guardian_approval", 独立 subagent 检视每个请求, 显著的额外 token 消耗
approvals_reviewer = "guardian_subagent"

[features]
# 使用多 agent 协作
multi_agent = true
# 高清图片, 可用于调整界面布局细节
image_detail_original = true
# 独立 subagent 检视每个请求
guardian_approval = true

给它全开了, approvals_reviewer我没试过, 其它的都开了, 体验提升还可以.
